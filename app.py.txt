from flask import Flask, request
from twilio.twiml.messaging_response import MessagingResponse

app = Flask(__name__)

@app.route("/whatsapp", methods=['POST'])
def whatsapp_reply():
    incoming_msg = request.form.get('Body')
    resp = MessagingResponse()
    msg = resp.message()

    if 'أرض' in incoming_msg:
        reply = "جميل! ممكن تخبرني مساحة الأرض وموقعها؟"
    elif 'تصميم' in incoming_msg:
        reply = "هل عندك تصميم مبدئي أو حابب نبدأ لك من الصفر؟"
    else:
        reply = "مرحباً 👋 معك الرؤية المتكاملة للاستشارات الهندسية. كيف أقدر أساعدك اليوم؟"

    msg.body(reply)
    return str(resp)

if __name__ == "__main__":
    app.run(debug=True)
